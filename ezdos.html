Server target untuk pengujian DDoS — jalankan dengan Python

1. Buat file server.py dengan kode di bawah ini:

#!/usr/bin/env python3 import http.server import socketserver import json import time from collections import defaultdict PORT = 8080 STATS = { 'total': 0, 'success': 0, 'failed': 0, 'requests': [], # simpan 200 log terakhir 'start_time': time.time() } RATE_WINDOW = 60 # detik class CustomHandler(http.server.SimpleHTTPRequestHandler): def log_request(self, code='-', size='-'): # override untuk mencatat request pass def send_response(self, code, message=None): super().send_response(code, message) # catat statistik STATS['total'] += 1 if 200 <= code < 300: STATS['success'] += 1 else: STATS['failed'] += 1 # simpan log (max 200) STATS['requests'].append({ 'time': time.strftime('%H:%M:%S'), 'status': code, 'path': self.path, 'ip': self.client_address[0] }) if len(STATS['requests']) > 200: STATS['requests'] = STATS['requests'][-200:] def do_GET(self): if self.path == '/stats': self.send_response(200) self.send_header('Content-Type', 'application/json') self.end_headers() # hitung rate now = time.time() recent = [r for r in STATS['requests'] if now - r.get('_ts', 0) < RATE_WINDOW] rate = len(recent) / RATE_WINDOW data = { 'total': STATS['total'], 'success': STATS['success'], 'failed': STATS['failed'], 'rate': round(rate, 2), 'requests': STATS['requests'][-50:], 'uptime': int(now - STATS['start_time']) } self.wfile.write(json.dumps(data).encode()) return # serve file statis (termasuk halaman ini) return super().do_GET() def log_message(self, format, *args): # matikan logging bawaan pass Handler = CustomHandler with socketserver.TCPServer(("0.0.0.0", PORT), Handler) as httpd: print(f"✅ Server target berjalan di http://localhost:{PORT}") print("🚀 Tekan Ctrl+C untuk berhenti") try: httpd.serve_forever() except KeyboardInterrupt: print("\n⏹ Server dihentikan.")
atau download langsung

2. Jalankan server:

python3 server.py

Server akan berjalan di http://localhost:8080

Setelah server berjalan, buka ulang halaman ini (atau akses http://localhost:8080) untuk melihat dashboard target.

Hanya untuk pengujian di server sendiri.
